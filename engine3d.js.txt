/*!
 * BV3D — BuildVerse mini 3D engine (WebGL2, zero dependencies)
 * Team The Game Changers · Hack Day S01 Genesis
 *
 * Features: boxes / cylinders / cones / spheres / gable prisms, sun + sky light,
 * soft shadow maps, distance fog, gradient sky with sun glow, procedural lit windows,
 * ACES tone mapping, parent/child groups, orbit + pinch controls with tap detection.
 * Works offline (no CDN) and from file://.
 */
(function (global) {
  'use strict';

  /* ---------------- math (column-major 4x4) ---------------- */
  function m4() { const m = new Float32Array(16); m[0] = m[5] = m[10] = m[15] = 1; return m; }

  function mul(o, a, b) {
    const a00 = a[0], a01 = a[1], a02 = a[2], a03 = a[3], a10 = a[4], a11 = a[5], a12 = a[6], a13 = a[7],
      a20 = a[8], a21 = a[9], a22 = a[10], a23 = a[11], a30 = a[12], a31 = a[13], a32 = a[14], a33 = a[15];
    for (let i = 0; i < 4; i++) {
      const b0 = b[i * 4], b1 = b[i * 4 + 1], b2 = b[i * 4 + 2], b3 = b[i * 4 + 3];
      o[i * 4] = b0 * a00 + b1 * a10 + b2 * a20 + b3 * a30;
      o[i * 4 + 1] = b0 * a01 + b1 * a11 + b2 * a21 + b3 * a31;
      o[i * 4 + 2] = b0 * a02 + b1 * a12 + b2 * a22 + b3 * a32;
      o[i * 4 + 3] = b0 * a03 + b1 * a13 + b2 * a23 + b3 * a33;
    }
    return o;
  }

  function perspective(o, fovy, aspect, near, far) {
    const f = 1 / Math.tan(fovy / 2), nf = 1 / (near - far);
    o.fill(0); o[0] = f / aspect; o[5] = f; o[10] = (far + near) * nf; o[11] = -1; o[14] = 2 * far * near * nf;
    return o;
  }

  function ortho(o, l, r, b, t, n, f) {
    const lr = 1 / (l - r), bt = 1 / (b - t), nf = 1 / (n - f);
    o.fill(0); o[0] = -2 * lr; o[5] = -2 * bt; o[10] = 2 * nf;
    o[12] = (l + r) * lr; o[13] = (t + b) * bt; o[14] = (f + n) * nf; o[15] = 1;
    return o;
  }

  function lookAt(o, eye, c, up) {
    let z0 = eye[0] - c[0], z1 = eye[1] - c[1], z2 = eye[2] - c[2];
    let len = 1 / (Math.hypot(z0, z1, z2) || 1); z0 *= len; z1 *= len; z2 *= len;
    let x0 = up[1] * z2 - up[2] * z1, x1 = up[2] * z0 - up[0] * z2, x2 = up[0] * z1 - up[1] * z0;
    len = Math.hypot(x0, x1, x2);
    if (!len) { x0 = 1; x1 = 0; x2 = 0; } else { len = 1 / len; x0 *= len; x1 *= len; x2 *= len; }
    const y0 = z1 * x2 - z2 * x1, y1 = z2 * x0 - z0 * x2, y2 = z0 * x1 - z1 * x0;
    o[0] = x0; o[1] = y0; o[2] = z0; o[3] = 0;
    o[4] = x1; o[5] = y1; o[6] = z1; o[7] = 0;
    o[8] = x2; o[9] = y2; o[10] = z2; o[11] = 0;
    o[12] = -(x0 * eye[0] + x1 * eye[1] + x2 * eye[2]);
    o[13] = -(y0 * eye[0] + y1 * eye[1] + y2 * eye[2]);
    o[14] = -(z0 * eye[0] + z1 * eye[1] + z2 * eye[2]);
    o[15] = 1;
    return o;
  }

  // T * Ry * Rx * Rz
  function compose(o, p, r) {
    const cy = Math.cos(r[1]), sy = Math.sin(r[1]), cx = Math.cos(r[0]), sx = Math.sin(r[0]), cz = Math.cos(r[2]), sz = Math.sin(r[2]);
    o[0] = cy * cz + sy * sx * sz; o[1] = cx * sz; o[2] = -sy * cz + cy * sx * sz; o[3] = 0;
    o[4] = -cy * sz + sy * sx * cz; o[5] = cx * cz; o[6] = sy * sz + cy * sx * cz; o[7] = 0;
    o[8] = sy * cx; o[9] = -sx; o[10] = cy * cx; o[11] = 0;
    o[12] = p[0]; o[13] = p[1]; o[14] = p[2]; o[15] = 1;
    return o;
  }

  // 6 frustum planes from a view-projection matrix (column-major)
  function planes(m, out) {
    const r = (i, j) => m[j * 4 + i];
    const P = [[3, 0, 1], [3, 0, -1], [3, 1, 1], [3, 1, -1], [3, 2, 1], [3, 2, -1]];
    for (let k = 0; k < 6; k++) {
      const [w, a, sg] = P[k];
      const x = r(w, 0) + sg * r(a, 0), y = r(w, 1) + sg * r(a, 1), z = r(w, 2) + sg * r(a, 2), d = r(w, 3) + sg * r(a, 3);
      const l = Math.hypot(x, y, z) || 1; out[k * 4] = x / l; out[k * 4 + 1] = y / l; out[k * 4 + 2] = z / l; out[k * 4 + 3] = d / l;
    }
    return out;
  }
  function inFrustum(pl, o) {
    const w = o._w, rad = 0.5 * Math.hypot(o.scale[0], o.scale[1], o.scale[2]);
    for (let k = 0; k < 6; k++) if (pl[k * 4] * w[12] + pl[k * 4 + 1] * w[13] + pl[k * 4 + 2] * w[14] + pl[k * 4 + 3] < -rad) return false;
    return true;
  }

  function norm3(v) { const l = Math.hypot(v[0], v[1], v[2]) || 1; return [v[0] / l, v[1] / l, v[2] / l]; }

  function hexToLinear(hex) {
    if (Array.isArray(hex)) return hex.slice();
    let h = String(hex).replace('#', '');
    if (h.length === 3) h = h.split('').map(c => c + c).join('');
    const n = parseInt(h, 16);
    return [(n >> 16) & 255, (n >> 8) & 255, n & 255].map(c => Math.pow(c / 255, 2.2));
  }

  /* ---------------- geometry ---------------- */
  function G() { return { p: [], n: [], i: [] }; }
  function quad(g, a, b, c, d, n) {
    const s = g.p.length / 3;
    g.p.push(...a, ...b, ...c, ...d); for (let k = 0; k < 4; k++) g.n.push(...n);
    g.i.push(s, s + 1, s + 2, s, s + 2, s + 3);
  }
  function tri(g, a, b, c, n) {
    const s = g.p.length / 3;
    g.p.push(...a, ...b, ...c); for (let k = 0; k < 3; k++) g.n.push(...n);
    g.i.push(s, s + 1, s + 2);
  }
  function boxGeo() {
    const g = G(), h = 0.5;
    quad(g, [h, -h, -h], [h, h, -h], [h, h, h], [h, -h, h], [1, 0, 0]);
    quad(g, [-h, -h, h], [-h, h, h], [-h, h, -h], [-h, -h, -h], [-1, 0, 0]);
    quad(g, [-h, h, -h], [-h, h, h], [h, h, h], [h, h, -h], [0, 1, 0]);
    quad(g, [-h, -h, h], [-h, -h, -h], [h, -h, -h], [h, -h, h], [0, -1, 0]);
    quad(g, [-h, -h, h], [h, -h, h], [h, h, h], [-h, h, h], [0, 0, 1]);
    quad(g, [h, -h, -h], [-h, -h, -h], [-h, h, -h], [h, h, -h], [0, 0, -1]);
    return g;
  }
  function cylGeo(seg) {
    const g = G();
    for (let s = 0; s <= seg; s++) {
      const a = s / seg * Math.PI * 2, c = Math.cos(a), si = Math.sin(a);
      g.p.push(c * 0.5, -0.5, si * 0.5, c * 0.5, 0.5, si * 0.5); g.n.push(c, 0, si, c, 0, si);
    }
    for (let s = 0; s < seg; s++) { const k = s * 2; g.i.push(k, k + 1, k + 3, k, k + 3, k + 2); }
    [-0.5, 0.5].forEach(y => {
      const base = g.p.length / 3; g.p.push(0, y, 0); g.n.push(0, Math.sign(y), 0);
      for (let s = 0; s <= seg; s++) { const a = s / seg * Math.PI * 2; g.p.push(Math.cos(a) * 0.5, y, Math.sin(a) * 0.5); g.n.push(0, Math.sign(y), 0); }
      for (let s = 0; s < seg; s++) g.i.push(base, base + 1 + s, base + 2 + s);
    });
    return g;
  }
  function coneGeo(seg) {
    const g = G();
    for (let s = 0; s <= seg; s++) {
      const a = s / seg * Math.PI * 2, c = Math.cos(a), si = Math.sin(a);
      const n = norm3([c, 0.5, si]);
      g.p.push(c * 0.5, -0.5, si * 0.5, 0, 0.5, 0); g.n.push(...n, ...n);
    }
    for (let s = 0; s < seg; s++) { const k = s * 2; g.i.push(k, k + 1, k + 2); }
    const base = g.p.length / 3; g.p.push(0, -0.5, 0); g.n.push(0, -1, 0);
    for (let s = 0; s <= seg; s++) { const a = s / seg * Math.PI * 2; g.p.push(Math.cos(a) * 0.5, -0.5, Math.sin(a) * 0.5); g.n.push(0, -1, 0); }
    for (let s = 0; s < seg; s++) g.i.push(base, base + 1 + s, base + 2 + s);
    return g;
  }
  function sphereGeo(lat, lon) {
    const g = G();
    for (let y = 0; y <= lat; y++) {
      const v = y / lat * Math.PI, sv = Math.sin(v), cv = Math.cos(v);
      for (let x = 0; x <= lon; x++) {
        const u = x / lon * Math.PI * 2, n = [Math.cos(u) * sv, cv, Math.sin(u) * sv];
        g.p.push(n[0] * 0.5, n[1] * 0.5, n[2] * 0.5); g.n.push(...n);
      }
    }
    for (let y = 0; y < lat; y++) for (let x = 0; x < lon; x++) {
      const a = y * (lon + 1) + x, b = a + lon + 1;
      g.i.push(a, b, a + 1, b, b + 1, a + 1);
    }
    return g;
  }
  // gable roof: ridge along z, triangle in x-y
  function prismGeo() {
    const g = G(), h = 0.5;
    const L = norm3([-1, 0.5, 0]), R = norm3([1, 0.5, 0]);
    quad(g, [-h, -h, -h], [-h, -h, h], [0, h, h], [0, h, -h], L);
    quad(g, [h, -h, h], [h, -h, -h], [0, h, -h], [0, h, h], R);
    quad(g, [-h, -h, -h], [h, -h, -h], [h, -h, h], [-h, -h, h], [0, -1, 0]);
    tri(g, [-h, -h, h], [h, -h, h], [0, h, h], [0, 0, 1]);
    tri(g, [h, -h, -h], [-h, -h, -h], [0, h, -h], [0, 0, -1]);
    return g;
  }

  /* ---------------- shaders ---------------- */
  const VS = `#version 300 es
  layout(location=0) in vec3 aPos; layout(location=1) in vec3 aNor;
  uniform mat4 uModel, uVP, uLightVP; uniform vec3 uScale;
  out vec3 vWorld; out vec3 vNor; out vec4 vLight; out vec3 vLocal;
  void main(){
    vec4 w = uModel * vec4(aPos*uScale, 1.0);
    vWorld = w.xyz; vLocal = aPos*uScale;
    vNor = mat3(uModel) * normalize(aNor / uScale);
    vLight = uLightVP * w;
    gl_Position = uVP * w;
  }`;

  const FS = `#version 300 es
  precision highp float; precision highp sampler2DShadow;
  in vec3 vWorld; in vec3 vNor; in vec4 vLight; in vec3 vLocal;
  uniform vec3 uColor, uEmissive, uSunDir, uSunColor, uSky, uGround, uFog, uCam;
  uniform float uAlpha, uSpec, uFogNear, uFogFar, uShadowOn, uExposure, uTexel;
  uniform float uWin, uWinLit, uWinSeed; uniform vec2 uWinSize; uniform vec3 uWinColor;
  uniform sampler2DShadow uShadow;
  out vec4 outColor;
  float hash(vec2 p){ return fract(sin(dot(p, vec2(127.1,311.7))) * 43758.5453); }
  float shadowF(){
    if (uShadowOn < 0.5) return 1.0;
    vec3 p = vLight.xyz / vLight.w * 0.5 + 0.5;
    if (p.x < 0.0 || p.x > 1.0 || p.y < 0.0 || p.y > 1.0 || p.z > 1.0) return 1.0;
    float s = 0.0;
    for (int x=-1; x<=1; x++) for (int y=-1; y<=1; y++)
      s += texture(uShadow, vec3(p.xy + vec2(float(x), float(y)) * uTexel * 1.25, p.z - 0.0012));
    return s / 9.0;
  }
  vec3 aces(vec3 x){ return clamp((x*(2.51*x+0.03))/(x*(2.43*x+0.59)+0.14), 0.0, 1.0); }
  void main(){
    vec3 n = normalize(vNor);
    vec3 base = uColor; vec3 emis = uEmissive; float spec = uSpec;
    if (uWin > 0.5 && abs(n.y) < 0.5) {
      float hc = (abs(n.x) > 0.5 ? vWorld.z : vWorld.x);
      vec2 uv = vec2(hc / uWinSize.x, vWorld.y / uWinSize.y);
      vec2 cell = floor(uv); vec2 f = fract(uv);
      float inW = step(0.18, f.x) * step(f.x, 0.82) * step(0.28, f.y) * step(f.y, 0.82);
      float h = hash(cell + vec2(uWinSeed, n.x*3.0 + n.z*7.0));
      float lit = step(1.0 - uWinLit, h);
      base = mix(base, vec3(0.02,0.035,0.05), inW);
      emis += inW * lit * uWinColor * (0.6 + 0.8*hash(cell.yx + 3.1));
      spec = mix(spec, 0.9, inW);
    }
    float ndl = max(dot(n, uSunDir), 0.0);
    float sh = shadowF();
    vec3 hemi = mix(uGround, uSky, n.y * 0.5 + 0.5);
    vec3 v = normalize(uCam - vWorld); vec3 hv = normalize(uSunDir + v);
    vec3 col = base * (hemi + uSunColor * ndl * sh) + uSunColor * pow(max(dot(n, hv), 0.0), 48.0) * spec * sh + emis;
    float d = length(uCam - vWorld);
    col = mix(col, uFog, smoothstep(uFogNear, uFogFar, d));
    col = aces(col * uExposure);
    outColor = vec4(pow(col, vec3(1.0/2.2)), uAlpha);
  }`;

  const SVS = `#version 300 es
  layout(location=0) in vec3 aPos; uniform mat4 uModel, uLightVP; uniform vec3 uScale;
  void main(){ gl_Position = uLightVP * uModel * vec4(aPos*uScale, 1.0); }`;
  const SFS = `#version 300 es
  precision mediump float; void main(){}`;

  const SKYVS = `#version 300 es
  out vec2 vNdc;
  void main(){ vec2 p = vec2(float((gl_VertexID<<1)&2), float(gl_VertexID&2)); vNdc = p*2.0-1.0; gl_Position = vec4(vNdc, 0.999, 1.0); }`;
  const SKYFS = `#version 300 es
  precision highp float; in vec2 vNdc;
  uniform vec3 uRight, uUp, uFwd, uTop, uHorizon, uGroundSky, uSunDir, uSunColor; uniform float uTan, uAspect, uExposure, uSunGlow;
  out vec4 outColor;
  vec3 aces(vec3 x){ return clamp((x*(2.51*x+0.03))/(x*(2.43*x+0.59)+0.14), 0.0, 1.0); }
  void main(){
    vec3 dir = normalize(uFwd + vNdc.x * uTan * uAspect * uRight + vNdc.y * uTan * uUp);
    float t = dir.y;
    vec3 c = t > 0.0 ? mix(uHorizon, uTop, pow(t, 0.55)) : mix(uHorizon, uGroundSky, pow(min(-t*4.0,1.0), 0.8));
    float sd = max(dot(dir, uSunDir), 0.0);
    c += uSunColor * (pow(sd, 900.0) * 6.0 + pow(sd, 12.0) * 0.18 * uSunGlow);
    c = aces(c * uExposure);
    outColor = vec4(pow(c, vec3(1.0/2.2)), 1.0);
  }`;

  function compile(gl, type, src) {
    const s = gl.createShader(type); gl.shaderSource(s, src); gl.compileShader(s);
    if (!gl.getShaderParameter(s, gl.COMPILE_STATUS)) throw new Error(gl.getShaderInfoLog(s));
    return s;
  }
  function program(gl, vs, fs) {
    const p = gl.createProgram();
    gl.attachShader(p, compile(gl, gl.VERTEX_SHADER, vs));
    gl.attachShader(p, compile(gl, gl.FRAGMENT_SHADER, fs));
    gl.linkProgram(p);
    if (!gl.getProgramParameter(p, gl.LINK_STATUS)) throw new Error(gl.getProgramInfoLog(p));
    const u = {}; const n = gl.getProgramParameter(p, gl.ACTIVE_UNIFORMS);
    for (let i = 0; i < n; i++) { const info = gl.getActiveUniform(p, i); u[info.name] = gl.getUniformLocation(p, info.name); }
    return { p, u };
  }
  function upload(gl, g) {
    const vao = gl.createVertexArray(); gl.bindVertexArray(vao);
    const pb = gl.createBuffer(); gl.bindBuffer(gl.ARRAY_BUFFER, pb); gl.bufferData(gl.ARRAY_BUFFER, new Float32Array(g.p), gl.STATIC_DRAW);
    gl.enableVertexAttribArray(0); gl.vertexAttribPointer(0, 3, gl.FLOAT, false, 0, 0);
    const nb = gl.createBuffer(); gl.bindBuffer(gl.ARRAY_BUFFER, nb); gl.bufferData(gl.ARRAY_BUFFER, new Float32Array(g.n), gl.STATIC_DRAW);
    gl.enableVertexAttribArray(1); gl.vertexAttribPointer(1, 3, gl.FLOAT, false, 0, 0);
    const ib = gl.createBuffer(); gl.bindBuffer(gl.ELEMENT_ARRAY_BUFFER, ib); gl.bufferData(gl.ELEMENT_ARRAY_BUFFER, new Uint16Array(g.i), gl.STATIC_DRAW);
    gl.bindVertexArray(null);
    return { vao, count: g.i.length };
  }

  /* ---------------- objects ---------------- */
  function mesh(type, o) {
    o = o || {};
    return {
      type, pos: (o.pos || [0, 0, 0]).slice(), rot: (o.rot || [0, 0, 0]).slice(), scale: (o.scale || [1, 1, 1]).slice(),
      color: hexToLinear(o.color || '#cccccc'), alpha: o.alpha == null ? 1 : o.alpha,
      emissive: o.emissive ? hexToLinear(o.emissive).map(c => c * (o.emissiveStrength || 1)) : [0, 0, 0],
      spec: o.spec == null ? 0.15 : o.spec, win: o.win || null, parent: o.parent || null,
      visible: o.visible !== false, castShadow: o.castShadow !== false, tag: o.tag || null,
      _l: m4(), _w: m4(), _f: -1
    };
  }
  function group(o) {
    o = o || {};
    return { isGroup: true, pos: (o.pos || [0, 0, 0]).slice(), rot: (o.rot || [0, 0, 0]).slice(), parent: o.parent || null, visible: o.visible !== false, _l: m4(), _w: m4(), _f: -1 };
  }

  /* ---------------- engine ---------------- */
  function create(canvas, opts) {
    opts = opts || {};
    const gl = canvas.getContext('webgl2', { antialias: true, alpha: false, preserveDrawingBuffer: !!opts.preserve, powerPreference: 'high-performance' });
    if (!gl) throw new Error('WebGL2 is not available in this browser');
    const main = program(gl, VS, FS), shad = program(gl, SVS, SFS), sky = program(gl, SKYVS, SKYFS);
    const geos = { box: upload(gl, boxGeo()), cyl: upload(gl, cylGeo(20)), cone: upload(gl, coneGeo(20)), sphere: upload(gl, sphereGeo(10, 18)), prism: upload(gl, prismGeo()) };
    const skyVao = gl.createVertexArray();

    const SM = opts.shadowRes || 2048;
    const depthTex = gl.createTexture();
    gl.bindTexture(gl.TEXTURE_2D, depthTex);
    gl.texStorage2D(gl.TEXTURE_2D, 1, gl.DEPTH_COMPONENT24, SM, SM);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MIN_FILTER, gl.LINEAR);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MAG_FILTER, gl.LINEAR);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_S, gl.CLAMP_TO_EDGE);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_T, gl.CLAMP_TO_EDGE);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_COMPARE_MODE, gl.COMPARE_REF_TO_TEXTURE);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_COMPARE_FUNC, gl.LEQUAL);
    const fbo = gl.createFramebuffer();
    gl.bindFramebuffer(gl.FRAMEBUFFER, fbo);
    gl.framebufferTexture2D(gl.FRAMEBUFFER, gl.DEPTH_ATTACHMENT, gl.TEXTURE_2D, depthTex, 0);
    gl.drawBuffers([gl.NONE]); gl.readBuffer(gl.NONE);
    gl.bindFramebuffer(gl.FRAMEBUFFER, null);

    const proj = m4(), view = m4(), vp = m4(), lview = m4(), lproj = m4(), lvp = m4();
    const camPl = new Float32Array(24), litPl = new Float32Array(24);
    let frame = 0;

    const eng = {
      gl, canvas, objects: [],
      camera: { target: [0, 0, 0], dist: 60, theta: 0.7, phi: 1.05, fov: 45, near: 0.3, far: 3000, shake: [0, 0, 0], minPhi: 0.15, maxPhi: 1.52, minDist: 5, maxDist: 600 },
      sun: { dir: norm3([0.5, 0.8, 0.35]), color: hexToLinear('#fff1dc'), intensity: 3.0 },
      light: { sky: hexToLinear('#b9d4f2'), ground: hexToLinear('#6d6a5f'), skyI: 0.9, groundI: 0.6 },
      skyColors: { top: hexToLinear('#3f7cc4'), horizon: hexToLinear('#cfe1f2'), ground: hexToLinear('#8a9480'), glow: 1 },
      fog: { color: null, near: 150, far: 900 },
      shadow: { enabled: opts.shadows !== false, center: [0, 0, 0], size: 80 },
      exposure: 1.0,
      add(...objs) { objs.forEach(o => { if (o && !o.isGroup) this.objects.push(o); }); return objs[0]; },
      remove(o) { const i = this.objects.indexOf(o); if (i >= 0) this.objects.splice(i, 1); },
      clear() { this.objects.length = 0; },
      mesh(type, o) { return this.add(mesh(type, o)); },
      render,
      resize,
      eye: [0, 0, 0]
    };

    function world(o) {
      if (o._f === frame) return o._w;
      compose(o._l, o.pos, o.rot);
      if (o.parent) mul(o._w, world(o.parent), o._l); else o._w.set(o._l);
      o._f = frame; return o._w;
    }
    function shown(o) { while (o) { if (!o.visible) return false; o = o.parent; } return true; }

    function resize() {
      const dpr = Math.min(global.devicePixelRatio || 1, opts.maxDpr || 2);
      const w = Math.max(1, Math.round(canvas.clientWidth * dpr)), h = Math.max(1, Math.round(canvas.clientHeight * dpr));
      if (canvas.width !== w || canvas.height !== h) { canvas.width = w; canvas.height = h; }
    }

    function render() {
      resize(); frame++;
      const c = eng.camera, W = canvas.width, H = canvas.height, aspect = W / H;
      const t = c.target;
      const eye = [
        t[0] + c.dist * Math.sin(c.phi) * Math.sin(c.theta) + c.shake[0],
        t[1] + c.dist * Math.cos(c.phi) + c.shake[1],
        t[2] + c.dist * Math.sin(c.phi) * Math.cos(c.theta) + c.shake[2]];
      eng.eye = eye;
      const tgt = [t[0] + c.shake[0] * 0.6, t[1] + c.shake[1] * 0.6, t[2] + c.shake[2] * 0.6];
      perspective(proj, c.fov * Math.PI / 180, aspect, c.near, c.far);
      lookAt(view, eye, tgt, [0, 1, 0]);
      mul(vp, proj, view);

      const sd = norm3(eng.sun.dir), S = eng.shadow.size, sc = eng.shadow.center;
      const leye = [sc[0] + sd[0] * S * 1.5, sc[1] + sd[1] * S * 1.5, sc[2] + sd[2] * S * 1.5];
      lookAt(lview, leye, sc, Math.abs(sd[1]) > 0.99 ? [0, 0, 1] : [0, 1, 0]);
      ortho(lproj, -S, S, -S, S, 0.1, S * 3.2);
      mul(lvp, lproj, lview);

      const all = eng.objects.filter(shown);
      all.forEach(world);
      planes(vp, camPl); planes(lvp, litPl);
      const list = all.filter(o => inFrustum(camPl, o));
      const casters = all.filter(o => o.castShadow && o.alpha >= 0.99 && inFrustum(litPl, o));
      eng.stats = { objects: all.length, drawn: list.length, casters: casters.length };
      const sunOn = sd[1] > 0.02 && eng.shadow.enabled;

      // shadow pass
      gl.enable(gl.DEPTH_TEST);
      gl.disable(gl.CULL_FACE);
      if (sunOn) {
        gl.bindFramebuffer(gl.FRAMEBUFFER, fbo);
        gl.viewport(0, 0, SM, SM);
        gl.clear(gl.DEPTH_BUFFER_BIT);
        gl.enable(gl.POLYGON_OFFSET_FILL); gl.polygonOffset(2.0, 4.0);
        gl.useProgram(shad.p);
        gl.uniformMatrix4fv(shad.u.uLightVP, false, lvp);
        for (const o of casters) {
          const g = geos[o.type];
          gl.uniformMatrix4fv(shad.u.uModel, false, o._w);
          gl.uniform3fv(shad.u.uScale, o.scale);
          gl.bindVertexArray(g.vao); gl.drawElements(gl.TRIANGLES, g.count, gl.UNSIGNED_SHORT, 0);
        }
        gl.disable(gl.POLYGON_OFFSET_FILL);
        gl.bindFramebuffer(gl.FRAMEBUFFER, null);
      }

      gl.viewport(0, 0, W, H);
      gl.clearColor(0, 0, 0, 1);
      gl.clear(gl.COLOR_BUFFER_BIT | gl.DEPTH_BUFFER_BIT);

      // sky
      const fwd = norm3([tgt[0] - eye[0], tgt[1] - eye[1], tgt[2] - eye[2]]);
      const right = norm3([fwd[1] * 0 - fwd[2] * 1, fwd[2] * 0 - fwd[0] * 0, fwd[0] * 1 - fwd[1] * 0]);
      const up = [right[1] * fwd[2] - right[2] * fwd[1], right[2] * fwd[0] - right[0] * fwd[2], right[0] * fwd[1] - right[1] * fwd[0]];
      const sunCol = eng.sun.color.map(v => v * eng.sun.intensity * Math.max(0, Math.min(1, sd[1] * 6 + 0.2)));
      gl.depthMask(false);
      gl.useProgram(sky.p);
      gl.uniform3fv(sky.u.uRight, right); gl.uniform3fv(sky.u.uUp, up); gl.uniform3fv(sky.u.uFwd, fwd);
      gl.uniform3fv(sky.u.uTop, eng.skyColors.top); gl.uniform3fv(sky.u.uHorizon, eng.skyColors.horizon); gl.uniform3fv(sky.u.uGroundSky, eng.skyColors.ground);
      gl.uniform3fv(sky.u.uSunDir, sd); gl.uniform3fv(sky.u.uSunColor, eng.sun.color);
      gl.uniform1f(sky.u.uTan, Math.tan(c.fov * Math.PI / 360)); gl.uniform1f(sky.u.uAspect, aspect);
      gl.uniform1f(sky.u.uExposure, eng.exposure); gl.uniform1f(sky.u.uSunGlow, eng.skyColors.glow);
      gl.bindVertexArray(skyVao); gl.drawArrays(gl.TRIANGLES, 0, 3);
      gl.depthMask(true);

      // main pass
      gl.useProgram(main.p);
      const u = main.u;
      gl.uniformMatrix4fv(u.uVP, false, vp);
      gl.uniformMatrix4fv(u.uLightVP, false, lvp);
      gl.uniform3fv(u.uSunDir, sd);
      gl.uniform3fv(u.uSunColor, sunCol);
      gl.uniform3fv(u.uSky, eng.light.sky.map(v => v * eng.light.skyI));
      gl.uniform3fv(u.uGround, eng.light.ground.map(v => v * eng.light.groundI));
      gl.uniform3fv(u.uFog, eng.fog.color || eng.skyColors.horizon);
      gl.uniform1f(u.uFogNear, eng.fog.near); gl.uniform1f(u.uFogFar, eng.fog.far);
      gl.uniform3fv(u.uCam, eye);
      gl.uniform1f(u.uShadowOn, sunOn ? 1 : 0);
      gl.uniform1f(u.uTexel, 1 / SM);
      gl.uniform1f(u.uExposure, eng.exposure);
      gl.activeTexture(gl.TEXTURE0); gl.bindTexture(gl.TEXTURE_2D, depthTex); gl.uniform1i(u.uShadow, 0);

      const opaque = [], trans = [];
      for (const o of list) (o.alpha < 0.99 ? trans : opaque).push(o);
      const draw = o => {
        const g = geos[o.type];
        gl.uniformMatrix4fv(u.uModel, false, o._w);
        gl.uniform3fv(u.uScale, o.scale);
        gl.uniform3fv(u.uColor, o.color);
        gl.uniform3fv(u.uEmissive, o.emissive);
        gl.uniform1f(u.uAlpha, o.alpha);
        gl.uniform1f(u.uSpec, o.spec);
        if (o.win) {
          gl.uniform1f(u.uWin, 1); gl.uniform2fv(u.uWinSize, o.win.size || [3, 3]);
          gl.uniform1f(u.uWinLit, o.win.lit == null ? 0.5 : o.win.lit);
          gl.uniform1f(u.uWinSeed, o.win.seed || 0);
          gl.uniform3fv(u.uWinColor, o.win.color || [0, 0, 0]);
        } else gl.uniform1f(u.uWin, 0);
        gl.bindVertexArray(g.vao); gl.drawElements(gl.TRIANGLES, g.count, gl.UNSIGNED_SHORT, 0);
      };
      opaque.forEach(draw);
      if (trans.length) {
        trans.sort((a, b) => {
          const da = (a._w[12] - eye[0]) ** 2 + (a._w[13] - eye[1]) ** 2 + (a._w[14] - eye[2]) ** 2;
          const db = (b._w[12] - eye[0]) ** 2 + (b._w[13] - eye[1]) ** 2 + (b._w[14] - eye[2]) ** 2;
          return db - da;
        });
        gl.enable(gl.BLEND); gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA); gl.depthMask(false);
        trans.forEach(draw);
        gl.depthMask(true); gl.disable(gl.BLEND);
      }
      gl.bindVertexArray(null);
    }

    return eng;
  }

  /* ---------------- orbit / pinch controls with tap detection ---------------- */
  function orbit(eng, opts) {
    opts = opts || {};
    const el = eng.canvas, cam = eng.camera, pts = new Map();
    let moved = 0, downAt = 0, lastPinch = 0;
    el.style.touchAction = 'none';
    el.addEventListener('pointerdown', e => {
      el.setPointerCapture(e.pointerId);
      pts.set(e.pointerId, { x: e.clientX, y: e.clientY });
      if (pts.size === 1) { moved = 0; downAt = performance.now(); }
      if (pts.size === 2) { const [a, b] = [...pts.values()]; lastPinch = Math.hypot(a.x - b.x, a.y - b.y); moved = 99; }
      opts.onInteract && opts.onInteract();
    });
    el.addEventListener('pointermove', e => {
      const p = pts.get(e.pointerId); if (!p) return;
      const dx = e.clientX - p.x, dy = e.clientY - p.y;
      p.x = e.clientX; p.y = e.clientY;
      if (pts.size === 1) {
        moved += Math.abs(dx) + Math.abs(dy);
        if (moved > 6 && opts.rotate !== false) {
          cam.theta -= dx * 0.008;
          cam.phi = Math.min(cam.maxPhi, Math.max(cam.minPhi, cam.phi - dy * 0.006));
        }
      } else if (pts.size === 2) {
        const [a, b] = [...pts.values()]; const d = Math.hypot(a.x - b.x, a.y - b.y);
        if (lastPinch) cam.dist = Math.min(cam.maxDist, Math.max(cam.minDist, cam.dist * lastPinch / d));
        lastPinch = d;
      }
    });
    const up = e => {
      if (!pts.has(e.pointerId)) return;
      pts.delete(e.pointerId);
      if (pts.size === 0 && moved <= 6 && performance.now() - downAt < 450 && opts.onTap) opts.onTap(e);
      if (pts.size < 2) lastPinch = 0;
    };
    el.addEventListener('pointerup', up);
    el.addEventListener('pointercancel', e => { pts.delete(e.pointerId); lastPinch = 0; });
    el.addEventListener('wheel', e => {
      e.preventDefault();
      cam.dist = Math.min(cam.maxDist, Math.max(cam.minDist, cam.dist * (1 + Math.sign(e.deltaY) * 0.08)));
      opts.onInteract && opts.onInteract();
    }, { passive: false });
  }

  global.BV3D = { create, orbit, mesh, group, color: hexToLinear, norm3 };
})(typeof window !== 'undefined' ? window : globalThis);
